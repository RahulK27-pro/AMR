python
import os
from launch import LaunchDescription
from launch.actions import ExecuteProcess, TimerAction
from launch_ros.actions import Node

def generate_launch_description():

    home_dir    = os.environ['HOME']
    pkg_path    = os.path.join(home_dir, 'agv_ws', 'src', 'agv_description')
    urdf_file   = os.path.join(pkg_path, 'urdf',   'agv.urdf')
    bridge_file = os.path.join(pkg_path, 'config', 'bridge.yaml')
    world_file  = os.path.join(pkg_path, 'worlds', 'warehouse.sdf')
    ekf_file    = os.path.join(pkg_path, 'config', 'ekf.yaml')   # ← new

    with open(urdf_file, 'r') as f:
        robot_desc = f.read()

    # A. Blueprint
    rsp_node = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        name='robot_state_publisher',
        parameters=[{
            'robot_description': robot_desc,
            'use_sim_time': True,
        }],
        output='screen'
    )

    # B. Physics engine
    gazebo = ExecuteProcess(
        cmd=['gz', 'sim', world_file],
        output='screen'
    )

    # C. Spawner
    spawn_node = TimerAction(
        period=2.0,
        actions=[
            Node(
                package='ros_gz_sim',
                executable='create',
                arguments=[
                    '-topic', 'robot_description',
                    '-name',  'warehouse_agv',
                    '-x',     '-2.0',
                    '-y',     '0.0',
                    '-z',     '0.09',
                    '-Y',     '0.0',
                ],
                output='screen'
            )
        ]
    )

    # D. Bridge
    bridge_node = Node(
        package='ros_gz_bridge',
        executable='parameter_bridge',
        name='ros_gz_bridge',
        arguments=['--ros-args', '-p', f'config_file:={bridge_file}'],
        output='screen'
    )

    # E. EKF — fuses /odom + /imu/data → /odometry/filtered
    #    This is now the authoritative pose estimate for the robot.
    #    Nav2 and SLAM will use /odometry/filtered, not raw /odom.
    ekf_node = Node(
        package='robot_localization',
        executable='ekf_node',
        name='ekf_filter_node',
        parameters=[
            ekf_file,
            {'use_sim_time': True}
        ],
        remappings=[
            # EKF publishes here — this becomes your clean odometry
            ('odometry/filtered', '/odometry/filtered'),
        ],
        output='screen'
    )

    # F. Static TF: map → odom (remove once SLAM is added)
    static_tf = Node(
        package='tf2_ros',
        executable='static_transform_publisher',
        name='static_tf_map_odom',
        arguments=['0', '0', '0', '0', '0', '0', 'map', 'odom'],
        parameters=[{'use_sim_time': True}],
        output='screen'
    )

    # G. Navigation
    tracer_node = Node(
        package='agv_navigation',
        executable='tracer',
        output='screen'
    )

    return LaunchDescription([
        gazebo,
        rsp_node,
        bridge_node,
        static_tf,
        ekf_node,        # ← added
        spawn_node,
        tracer_node,
    ])

Verify RViz Map QoS (Crucial)
Because maps are massive data files, SLAM Toolbox does not spam them over the network continuously. It publishes them occasionally, and "pins" them to the ROS 2 network using a specific durability profile.
If RViz is not configured to look for "pinned" messages, it will sit there completely blind, waiting for a fresh broadcast.

In RViz, expand the Map display on the left.
Expand the Topic section.
Change the Durability Policy from Volatile to Transient Local.
Change the Reliability Policy from Best Effort to Reliable.
The "Kickstart" Move SLAM algorithms are strictly motion-based. When they first boot up, they often will not publish the very first /map message until the robot has moved slightly and proven that its odometry is working. In your teleop terminal, drive the robot forward just a few centimeters. The moment the odometry registers movement, the SLAM solver will lock in its first calculation, and the gray grid with black walls will instantly pop onto your RViz screen. import os
from launch import LaunchDescription
from launch.actions import TimerAction
from launch_ros.actions import Node

def generate_launch_description():

    home_dir = os.environ['HOME']
    pkg_path = os.path.join(home_dir, 'agv_ws', 'src', 'agv_description')

    # Your custom SLAM params file
    slam_params = os.path.join(pkg_path, 'config', 'mapper_params.yaml')

    # ─────────────────────────────────────────────
    # SLAM Toolbox node
    # Delayed 5s to ensure:
    #   - Gazebo is fully running
    #   - Robot is spawned
    #   - Bridge is publishing /scan and /tf
    #   - EKF is publishing /odometry/filtered
    # If SLAM starts before any of these are ready
    # it silently fails and never builds the map.
    # ─────────────────────────────────────────────
    slam_node = TimerAction(
        period=5.0,
        actions=[
            Node(
                package='slam_toolbox',
                executable='async_slam_toolbox_node',
                name='slam_toolbox',
                parameters=[
                    slam_params,
                    {'use_sim_time': True}
                ],
                output='screen'
            )
        ]
    )

    return LaunchDescription([
        slam_node,
    ])

import os
from launch import LaunchDescription
from launch.actions import (
    ExecuteProcess,
    RegisterEventHandler,
    TimerAction,
    LogInfo,
)
from launch.event_handlers import OnProcessStart, OnProcessIO
from launch_ros.actions import Node

def generate_launch_description():

    home_dir    = os.environ['HOME']
    pkg_path    = os.path.join(home_dir, 'agv_ws', 'src', 'agv_description')
    urdf_file   = os.path.join(pkg_path, 'urdf',   'agv.urdf')
    bridge_file = os.path.join(pkg_path, 'config', 'bridge.yaml')
    world_file  = os.path.join(pkg_path, 'worlds', 'warehouse.sdf')
    ekf_file    = os.path.join(pkg_path, 'config', 'ekf.yaml')

    with open(urdf_file, 'r') as f:
        robot_desc = f.read()

    # ─────────────────────────────────────────
    # STEP 1 — Gazebo
    # -r  = run immediately, not paused
    #        (paused = clock frozen = all sensor
    #         timestamps stuck at 0 = SLAM fails)
    # -v 2 = reduced verbosity, hides render spam
    # ─────────────────────────────────────────
    gazebo = ExecuteProcess(
        cmd=['gz', 'sim', '-r', '-v', '2', world_file],
        output='screen',
        name='gz'
    )

    # ─────────────────────────────────────────
    # STEP 2 — Bridge
    # Starts 3s after Gazebo process starts.
    # Not when Gazebo is "ready" (we can't detect
    # that) but 3s gives it time to load the world.
    # Starting bridge too early causes the
    # connect/disconnect loop that makes Gazebo twitch.
    # ─────────────────────────────────────────
    bridge_node = Node(
        package='ros_gz_bridge',
        executable='parameter_bridge',
        name='ros_gz_bridge',
        arguments=['--ros-args', '-p', f'config_file:={bridge_file}'],
        output='screen'
    )

    start_bridge_after_gazebo = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                LogInfo(msg='[1/5] Gazebo started. Waiting 3s for world to load...'),
                TimerAction(
                    period=3.0,
                    actions=[
                        LogInfo(msg='[2/5] Starting bridge...'),
                        bridge_node,
                    ]
                )
            ]
        )
    )

    # ─────────────────────────────────────────
    # STEP 3 — Robot State Publisher
    # Starts 5s after Gazebo (2s after bridge).
    # Bridge must be running so /clock is flowing
    # before RSP subscribes to it.
    # ─────────────────────────────────────────
    rsp_node = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        name='robot_state_publisher',
        parameters=[{
            'robot_description': robot_desc,
            'use_sim_time': True,
        }],
        output='screen'
    )

    start_rsp_after_gazebo = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=5.0,
                    actions=[
                        LogInfo(msg='[3/5] Starting robot_state_publisher...'),
                        rsp_node,
                    ]
                )
            ]
        )
    )

    # ─────────────────────────────────────────
    # STEP 4 — Spawn robot + EKF
    # Starts 8s after Gazebo.
    # RSP must be running before spawn so that
    # /robot_description topic exists for the spawner.
    # ─────────────────────────────────────────
    spawn_node = Node(
        package='ros_gz_sim',
        executable='create',
        arguments=[
            '-topic', 'robot_description',
            '-name',  'warehouse_agv',
            '-x',     '-2.0',
            '-y',     '0.0',
            '-z',     '0.09',
            '-Y',     '0.0',
        ],
        output='screen'
    )

    ekf_node = Node(
        package='robot_localization',
        executable='ekf_node',
        name='ekf_filter_node',
        parameters=[
            ekf_file,
            {
                'use_sim_time': True,
                'frequency': 30.0,
            }
        ],
        remappings=[
            ('odometry/filtered', '/odometry/filtered'),
        ],
        output='screen'
    )

    start_robot_after_gazebo = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=8.0,
                    actions=[
                        LogInfo(msg='[4/5] Spawning robot and starting EKF...'),
                        spawn_node,
                        ekf_node,
                    ]
                )
            ]
        )
    )

    # ─────────────────────────────────────────
    # STEP 5 — Navigation node
    # Starts 10s after Gazebo.
    # Robot must exist before tracer tries to
    # read wheel encoders and IMU.
    # ─────────────────────────────────────────
    tracer_node = Node(
        package='agv_navigation',
        executable='tracer',
        output='screen'
    )

    start_tracer_after_gazebo = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=10.0,
                    actions=[
                        LogInfo(msg='[5/5] Starting navigation node...'),
                        tracer_node,
                    ]
                )
            ]
        )
    )

    return LaunchDescription([

        start_bridge_after_gazebo,
        start_rsp_after_gazebo,
        start_robot_after_gazebo,
        start_tracer_after_gazebo,

Looking at everything carefully, I can identify the exact remaining problem. SLAM is running but silently rejecting every scan because of two issues combined:

Issue 1 — /tf is throttled to 30Hz in your bridge. SLAM does a TF lookup at the exact timestamp of each scan. Throttling creates gaps in the TF history — if a scan arrives at t=100ms and the nearest TF is at t=90ms or t=133ms, the lookup fails silently and that scan is dropped entirely. /tf must flow at full rate, never throttled.

Issue 2 — /tf_static is not bridged at all. Your fixed joints (base_link→lidar_link, base_link→camera_link etc.) are published by RSP on ROS 2 /tf_static directly. But Gazebo also publishes some static model info on its own /tf_static internally. Without bridging it, SLAM occasionally loses the static transform chain.

import os
from launch import LaunchDescription
from launch.actions import (
    ExecuteProcess,
    RegisterEventHandler,
    TimerAction,
    LogInfo,
)
from launch.event_handlers import OnProcessStart
from launch_ros.actions import Node

def generate_launch_description():

    home_dir    = os.environ['HOME']
    pkg_path    = os.path.join(home_dir, 'agv_ws', 'src', 'agv_description')
    urdf_file   = os.path.join(pkg_path, 'urdf',   'agv.urdf')
    bridge_file = os.path.join(pkg_path, 'config', 'bridge.yaml')
    world_file  = os.path.join(pkg_path, 'worlds', 'warehouse.sdf')
    ekf_file    = os.path.join(pkg_path, 'config', 'ekf.yaml')

    with open(urdf_file, 'r') as f:
        robot_desc = f.read()

    # 1. Gazebo — -r = run immediately so clock ticks from t=0
    gazebo = ExecuteProcess(
        cmd=['gz', 'sim', '-r', '-v', '2', world_file],
        output='screen',
        name='gz'
    )

    # 2. Bridge — 4s after Gazebo starts.
    #    Enough time for the world to finish loading.
    #    Starting too early causes the connect/disconnect
    #    loop that makes Gazebo twitch and freeze.
    bridge_node = Node(
        package='ros_gz_bridge',
        executable='parameter_bridge',
        name='ros_gz_bridge',
        arguments=['--ros-args', '-p', f'config_file:={bridge_file}'],
        output='screen'
    )

    start_bridge = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                LogInfo(msg='[1/5] Gazebo started — waiting 4s for world to load...'),
                TimerAction(
                    period=4.0,
                    actions=[
                        LogInfo(msg='[2/5] Bridge starting...'),
                        bridge_node,
                    ]
                )
            ]
        )
    )

    # 3. Robot State Publisher — 6s after Gazebo.
    #    Bridge must be running and /clock must be
    #    flowing before RSP starts, otherwise RSP
    #    stamps all TF at t=0 and SLAM rejects scans.
    rsp_node = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        name='robot_state_publisher',
        parameters=[{
            'robot_description': robot_desc,
            'use_sim_time': True,
        }],
        output='screen'
    )

    start_rsp = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=6.0,
                    actions=[
                        LogInfo(msg='[3/5] robot_state_publisher starting...'),
                        rsp_node,
                    ]
                )
            ]
        )
    )

    # 4. Spawner + EKF — 9s after Gazebo.
    #    RSP must be running so /robot_description
    #    topic exists when spawner requests it.
    spawn_node = Node(
        package='ros_gz_sim',
        executable='create',
        arguments=[
            '-topic', 'robot_description',
            '-name',  'warehouse_agv',
            '-x',     '-2.0',
            '-y',     '0.0',
            '-z',     '0.09',
            '-Y',     '0.0',
        ],
        output='screen'
    )

    ekf_node = Node(
        package='robot_localization',
        executable='ekf_node',
        name='ekf_filter_node',
        parameters=[
            ekf_file,
            {
                'use_sim_time': True,
                'frequency': 30.0,
            }
        ],
        remappings=[
            ('odometry/filtered', '/odometry/filtered'),
        ],
        output='screen'
    )

    start_robot = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=9.0,
                    actions=[
                        LogInfo(msg='[4/5] Spawning robot + starting EKF...'),
                        spawn_node,
                        ekf_node,
                    ]
                )
            ]
        )
    )

    # 5. Navigation — 12s after Gazebo
    tracer_node = Node(
        package='agv_navigation',
        executable='tracer',
        output='screen'
    )

    start_tracer = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=12.0,
                    actions=[
                        LogInfo(msg='[5/5] Navigation node starting...'),
                        tracer_node,
                    ]
                )
            ]
        )
    )

    return LaunchDescription([
        gazebo,
        start_bridge,
        start_rsp,
        start_robot,
        start_tracer,
    ])

The root cause is now clear. You have two nodes fighting over the same TF transform.

pythonimport os
from launch import LaunchDescription
from launch.actions import (
    ExecuteProcess,
    RegisterEventHandler,
    TimerAction,
    LogInfo,
)
from launch.event_handlers import OnProcessStart
from launch_ros.actions import Node

def generate_launch_description():

    home_dir    = os.environ['HOME']
    pkg_path    = os.path.join(home_dir, 'agv_ws', 'src', 'agv_description')
    urdf_file   = os.path.join(pkg_path, 'urdf',   'agv.urdf')
    bridge_file = os.path.join(pkg_path, 'config', 'bridge.yaml')
    world_file  = os.path.join(pkg_path, 'worlds', 'warehouse.sdf')

    with open(urdf_file, 'r') as f:
        robot_desc = f.read()

    # 1. Gazebo — -r runs immediately so clock ticks from t=0
    gazebo = ExecuteProcess(
        cmd=['gz', 'sim', '-r', '-v', '2', world_file],
        output='screen',
        name='gz'
    )

    # 2. Bridge — 4s after Gazebo so world fully loads first
    bridge_node = Node(
        package='ros_gz_bridge',
        executable='parameter_bridge',
        name='ros_gz_bridge',
        arguments=['--ros-args', '-p', f'config_file:={bridge_file}'],
        output='screen'
    )

    start_bridge = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                LogInfo(msg='[1/4] Gazebo started — waiting 4s...'),
                TimerAction(
                    period=4.0,
                    actions=[
                        LogInfo(msg='[2/4] Bridge starting...'),
                        bridge_node,
                    ]
                )
            ]
        )
    )

    # 3. RSP — 6s after Gazebo so /clock is flowing first
    rsp_node = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        name='robot_state_publisher',
        parameters=[{
            'robot_description': robot_desc,
            'use_sim_time': True,
        }],
        output='screen'
    )

    start_rsp = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=6.0,
                    actions=[
                        LogInfo(msg='[3/4] robot_state_publisher starting...'),
                        rsp_node,
                    ]
                )
            ]
        )
    )

    # 4. Spawner — 9s after Gazebo so RSP is publishing
    #    /robot_description before spawner requests it
    spawn_node = Node(
        package='ros_gz_sim',
        executable='create',
        arguments=[
            '-topic', 'robot_description',
            '-name',  'warehouse_agv',
            '-x',     '-2.0',
            '-y',     '0.0',
            '-z',     '0.09',
            '-Y',     '0.0',
        ],
        output='screen'
    )

    tracer_node = Node(
        package='agv_navigation',
        executable='tracer',
        output='screen'
    )

    start_robot = RegisterEventHandler(
        OnProcessStart(
            target_action=gazebo,
            on_start=[
                TimerAction(
                    period=9.0,
                    actions=[
                        LogInfo(msg='[4/4] Spawning robot + navigation...'),
                        spawn_node,
                        tracer_node,
                    ]
                )
            ]
        )
    )

    return LaunchDescription([
        gazebo,
        start_bridge,
        start_rsp,
        start_robot,
    ])
